import asyncio import secrets import time from pathlib import Path from aiogram import Bot, Dispatcher, types, F from aiogram.filters import Command, CommandObject from aiogram.fsm.context import FSMContext from aiogram.fsm.storage.memory import MemoryStorage from sqlalchemy import select, update from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession from sqlalchemy.orm import sessionmaker, DeclarativeBase, Mapped, mapped_column # --- НАСТРОЙКИ --- BOT_TOKEN = "ВАШ_ТОКЕН_ЗДЕСЬ" # Вставьте сюда ваш токен от @BotFather DEFAULT_BET = 50 # Ставка по умолчанию SPAM_TIMEOUT = 3 # Секунды между вращениями DB_PATH = Path("slotbot.db") # Файл базы данных # --- БАЗА ДАННЫХ (SQLite + SQLAlchemy) --- class Base(DeclarativeBase): pass class User(Base): __tablename__ = "users" user_id: Mapped[int] = mapped_column(primary_key=True) balance: Mapped[int] = mapped_column(default=1000) # Стартовый баланс last_spin_ts: Mapped[float] = mapped_column(default=0.0) engine = create_async_engine(f"sqlite+aiosqlite:///{DB_PATH}", echo=False, future=True) AsyncSessionLocal = sessionmaker(engine, class_=AsyncSession, expire_on_commit=False) async def init_db(): async with engine.begin() as conn: await conn.run_sync(Base.metadata.create_all) # --- ЛОГИКА ИГРЫ --- SYMBOLS = ["🍒", "🍋", "🔔", "💎", "7️⃣"] WIN_TABLE = { "🍒": 5, "🍋": 10, "🔔": 20, "💎": 50, "7️⃣": 100 } def check_win(reels: list[str]) -> tuple[bool, int]: """Проверяет линию. Возвращает (выигрыш: да/нет, множитель).""" if reels.count(reels[0]) == 3: return True, WIN_TABLE.get(reels[0], 0) if reels.count(reels[0]) == 2 or reels.count(reels[1]) == 2: return True, 1 return False, 0 # --- ОБРАБОТЧИКИ КОМАНД --- async def cmd_start(message: types.Message, state: FSMContext): async with AsyncSessionLocal() as session: async with session.begin(): result = await session.execute(select(User).where(User.user_id == message.from_user.id)) user = result.scalar_one_or_none() if not user: user = User(user_id=message.from_user.id) session.add(user) await session.commit() await message.answer( f"Привет, {message.from_user.full_name}! 🎰\n\n" f"Я — бот-слот. Вся валюта виртуальная.\n" f"Ваш текущий баланс: {user.balance} монет.\n\n" f"Как играть:\n" f"Напишите /slots <ставка>, чтобы крутануть барабаны.\n" f"Если не указывать ставку, будет использовано значение по умолчанию ({DEFAULT_BET}).\n" f"Примеры: /slots или /slots 100" ) async def play_slots(message: types.Message, state: FSMContext, bet: int): current_time = time.time() user_id = message.from_user.id async with AsyncSessionLocal() as session: async with session.begin(): result = await session.execute(select(User).where(User.user_id == user_id)) user = result.scalar_one_or_none() if not user: await message.answer("Ошибка профиля. Пожалуйста, нажмите /start еще раз.") return # Защита от повторного нажатия (спам-фильтр) if current_time - user.last_spin_ts < SPAM_TIMEOUT: wait_sec = round(SPAM_TIMEOUT - (current_time - user.last_spin_ts)) await message.answer(f"❗ Подождите {wait_sec} секунду(ы) перед следующим вращением.") return if bet > user.balance: await message.answer(f"Недостаточно средств. Ваш баланс: {user.balance}.") return if bet <= 0: await message.answer("Ставка должна быть положительным числом.") return # Обновляем время последнего спина ДО отправки сообщения, чтобы защитить от дублей user.last_spin_ts = current_time # Вращение reel1 = secrets.choice(SYMBOLS) reel2 = secrets.choice(SYMBOLS) reel3 = secrets.choice(SYMBOLS) is_won, multiplier = check_win([reel1, reel2, reel3]) winnings = bet * multiplier new_balance = user.balance - bet + winnings # Записываем новый баланс await session.execute( update(User) .where(User.user_id == user_id) .values(balance=new_balance) ) await session.commit() # Формируем ответ line_str = f"[{reel1}] [{reel2}] [{reel3}]" status_emoji = "🎉" if is_won else "❌" win_text = f"Выигрыш: +{winnings}" if winnings > 0 else "" reply = ( f"{status_emoji} {line_str} {status_emoji}\n\n" f"Ставка: {bet}\n" f"{win_text}\n\n" f"Текущий баланс: {new_balance}" ) await message.answer(reply) # --- ЗАПУСК БОТА --- async def main(): await init_db() bot = Bot(token=BOT_TOKEN, parse_mode="HTML") dp = Dispatcher(storage=MemoryStorage()) dp.message.register(cmd_start, Command(commands={"start"})) @dp.message(Command(commands={"slots"})) async def slots_wrapper(message: types.Message, command: CommandObject, state: FSMContext): try: raw_args = command.args or str(DEFAULT_BET) # Извлекаем первое число из текста команды (защита от мусора вроде "/slots 50 abc") parts = raw_args.split() bet = abs(int(parts[0])) except (ValueError, TypeError, IndexError): await message.answer(f"Используйте команду так: /slots [число]. Например: /slots {DEFAULT_BET}") return await play_slots(message, state, bet) print("✅ Бот успешно запущен!") await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
